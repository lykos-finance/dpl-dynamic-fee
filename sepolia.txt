contract HookFactory {
    function deploy(bytes memory code, bytes32 salt) public returns (address addr) {
        assembly { addr := create2(0, add(code, 0x20), mload(code), salt) }
        require(addr != address(0), "Factory failed");
    }
}

contract UniversalRouter is IUnlockCallback {
    IPoolManager manager;
    address public owner;

    constructor(IPoolManager _manager) {
        manager = _manager;
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    function addLiquidity(PoolKey memory key, int24 tl, int24 tu, int256 amount) external {
        manager.unlock(abi.encode(uint8(1), key, tl, tu, amount, msg.sender));
    }
    function removeLiquidity(PoolKey memory key, int24 tl, int24 tu, int256 amount) external {
        manager.unlock(abi.encode(uint8(1), key, tl, tu, -amount, msg.sender));
    }
    function swap(PoolKey memory key, bool zfo, int256 amount) external {
        manager.unlock(abi.encode(uint8(2), key, zfo, amount, msg.sender));
    }

    function unlockCallback(bytes calldata data) external override returns (bytes memory) {
        uint8 action = abi.decode(data, (uint8));
        PoolKey memory key;
        BalanceDelta delta;
        address recipient;

        if (action == 1) {
            int24 tl; int24 tu; int256 amount;
            (, key, tl, tu, amount, recipient) = abi.decode(data, (uint8, PoolKey, int24, int24, int256, address));
            (delta, ) = manager.modifyLiquidity(key, IPoolManager.ModifyLiquidityParams({
                tickLower: tl, tickUpper: tu, liquidityDelta: amount, salt: bytes32(0)
            }), "");
        } else if (action == 2) {
            bool zfo; int256 amt;
            (, key, zfo, amt, recipient) = abi.decode(data, (uint8, PoolKey, bool, int256, address));
            uint160 limit = zfo ? 4295128739 + 1 : 1461446703485210103287273052203988822378723970342 - 1;
            delta = manager.swap(key, IPoolManager.SwapParams({
                zeroForOne: zfo, amountSpecified: amt, sqrtPriceLimitX96: limit
            }), "");
        }

        if (delta.amount0() < 0)
            _pay(Currency.unwrap(key.currency0), uint256(uint128(-delta.amount0())));
        else if (delta.amount0() > 0)
            manager.take(key.currency0, address(this), uint256(uint128(delta.amount0())));

        if (delta.amount1() < 0)
            _pay(Currency.unwrap(key.currency1), uint256(uint128(-delta.amount1())));
        else if (delta.amount1() > 0)
            manager.take(key.currency1, address(this), uint256(uint128(delta.amount1())));

        return "";
    }

    // sync(address)   = 0xa5841194
    // settle()        = 0x11da60b4
    bytes4 private constant SYNC_SELECTOR   = bytes4(keccak256("sync(address)"));
    bytes4 private constant SETTLE_SELECTOR = bytes4(keccak256("settle()"));

    function _pay(address t, uint256 a) internal {
        (bool s1, ) = address(manager).call(abi.encodeWithSelector(SYNC_SELECTOR, t));
        require(s1, "Sync fail");
        _safeTransfer(t, address(manager), a);
        (bool s2, ) = address(manager).call(abi.encodeWithSelector(SETTLE_SELECTOR));
        require(s2, "Settle fail");
    }

    function _safeTransfer(address token, address to, uint256 amount) internal {
        (bool ok, bytes memory data) = token.call(
            abi.encodeWithSignature("transfer(address,uint256)", to, amount)
        );
        require(ok, "TRANSFER_FAILED: call reverted");
        if (data.length > 0) require(abi.decode(data, (bool)), "TRANSFER_FAILED: returned false");
    }

    function withdrawToken(address token, address to, uint256 amount) external onlyOwner {
        uint256 toWithdraw = amount;
        if (amount == type(uint256).max) {
            (bool okBal, bytes memory balData) = token.staticcall(
                abi.encodeWithSignature("balanceOf(address)", address(this))
            );
            require(okBal && balData.length >= 32, "Balance read failed");
            toWithdraw = abi.decode(balData, (uint256));
        }

        require(toWithdraw > 0, "Zero amount");
        _safeTransfer(token, to, toWithdraw);
    }
}

contract FullEndToEnd is Script {
    using CurrencyLibrary for Currency;

    address constant POOL_MANAGER = 0x000000000004444c5dc75cB358380D2e3dE08A90;
    address constant WETH         = 0xC02aaA39b223FE8D0A0e5C4F27eAD9083C756Cc2;
    address constant USDT         = 0xdAC17F958D2ee523a2206206994597C13D831ec7;

    function run() public {
        uint256 pk       = vm.envUint("PRIVATE_KEY");
        address deployer = vm.addr(pk);
        vm.startBroadcast(deployer);

        IPoolManager manager = IPoolManager(POOL_MANAGER);

        // ── 1. Deploy factory + hook ──
        HookFactory factory = new HookFactory();
        bytes memory code = abi.encodePacked(
            type(VolatilityGuardHookV2).creationCode,
            abi.encode(POOL_MANAGER)
        );

        vm.pauseGasMetering();
        uint160 target  = uint160(Hooks.BEFORE_SWAP_FLAG);
        uint160 mask    = 0x3FFF;
        uint256 saltInt = uint256(keccak256(abi.encodePacked(deployer, vm.unixTime())));
        address hookAddress;
        for (uint256 i = 0; i < 500_000; i++) {
            bytes32 salt = bytes32(saltInt + i);
            bytes32 hash = keccak256(abi.encodePacked(bytes1(0xff), address(factory), salt, keccak256(code)));
            address addr = address(uint160(uint256(hash)));
            if ((uint160(addr) & mask) == target) { hookAddress = addr; saltInt += i; break; }
        }
        vm.resumeGasMetering();
        require(hookAddress != address(0), "Hook mining failed");
        console.log("Hook:", hookAddress);
        factory.deploy(code, bytes32(saltInt));

        // ── 2. Sort tokens ──
        IERC20Like tokenA = IERC20Like(WETH);
        IERC20Like tokenB = IERC20Like(USDT);
        if (address(tokenA) > address(tokenB)) (tokenA, tokenB) = (tokenB, tokenA);
        console.log("Token0:", tokenA.symbol(), address(tokenA));
        console.log("Token1:", tokenB.symbol(), address(tokenB));

        // ── 3. Chainlink price ──
        (, int256 ethAnswer, , uint256 updatedAt, ) = AggregatorV3Interface(
            0x5f4eC3Df9cbd43714FE2740f5E3616155c5b8419
        ).latestRoundData();
        require(ethAnswer > 0, "Chainlink: invalid price");
        require(block.timestamp - updatedAt < 3600, "Chainlink: stale price");
        uint256 ethPriceUSD8 = uint256(ethAnswer);
        console.log("ETH/USD (x1e8):", ethPriceUSD8);

        // ── 4. Compute tick + sqrtPrice ──
        uint160 startSqrtPrice = uint160(sqrtU256(ethPriceUSD8) * (1 << 96) / 1e10);
        int24 currentTick = getTickFromSqrtPrice(startSqrtPrice);
        currentTick       = currentTick / 60 * 60;
        startSqrtPrice    = getSqrtRatioAtTick(currentTick);
        console.log("Current tick:", uint256(int256(currentTick)));

        // ── 5. Deploy router ──
        UniversalRouter router = new UniversalRouter(manager);

        // ── 6. Tick range ≈ ±5% от текущей цены ETH ──
        // 5% по цене соответствует ~487.9 тиков, округляем к spacing=60 => 480
        int24 tickDelta5Pct = 480;
        int24 tickLower = (currentTick - tickDelta5Pct) / 60 * 60;
        int24 tickUpper = (currentTick + tickDelta5Pct) / 60 * 60;

        uint160 sqrtLower   = getSqrtRatioAtTick(tickLower);
        uint160 sqrtCurrent = getSqrtRatioAtTick(currentTick);
        uint160 sqrtUpper   = getSqrtRatioAtTick(tickUpper);

        // ── 7. Compute L to use ~$1.5 of each token (~$3 total) ──
        // Target: 1.5 USDT + equivalent WETH
        uint256 Q96 = 2**96;
        uint256 targetUSDT = 1_000_000;  // 1 USDT
        uint256 targetWETH = (1_500_000 * 1e20) / ethPriceUSD8; // ~$1.5 in WETH
        
        // Compute L from both constraints, take minimum
        // L_usdt = Δtoken1 * 2^96 / (√upper - √current)
        uint128 liquidityFromUSDT = uint128(
            (targetUSDT * Q96) / (uint256(sqrtUpper) - uint256(sqrtCurrent))
        );
        // L_weth = Δtoken0 * √current * √lower / ((√current - √lower) * 2^96)
        uint128 liquidityFromWETH = uint128(
            (targetWETH * uint256(sqrtCurrent) * uint256(sqrtLower)) /
            (Q96 * (uint256(sqrtCurrent) - uint256(sqrtLower)))
        );
        
        uint128 liquidityAmount = liquidityFromWETH < liquidityFromUSDT 
            ? liquidityFromWETH 
            : liquidityFromUSDT;
        console.log("Liquidity (min of both):", liquidityAmount);
        console.log("  From USDT:", liquidityFromUSDT);
        console.log("  From WETH:", liquidityFromWETH);

        // ── 8. Compute token amounts needed for this L ──
        // Δtoken0 = L * (sqrtCurrent - sqrtLower) * 2^96 / (sqrtCurrent * sqrtLower)
        uint256 tokenASeed = (uint256(liquidityAmount) * Q96 *
            (uint256(sqrtCurrent) - uint256(sqrtLower))) /
            (uint256(sqrtCurrent) * uint256(sqrtLower));
        uint256 tokenBSeed = (uint256(liquidityAmount) *
            (uint256(sqrtUpper) - uint256(sqrtCurrent))) / Q96;

        console.log("Seed token0 (WETH):", tokenASeed);
        console.log("Seed token1 (USDT):", tokenBSeed);

        // Добавляем 1% запас
        tokenASeed = tokenASeed * 101 / 100;
        tokenBSeed = tokenBSeed * 101 / 100;

        require(tokenA.balanceOf(deployer) >= tokenASeed, "Not enough token0");
        require(tokenB.balanceOf(deployer) >= tokenBSeed, "Not enough token1");

        // ── 9. Fund router ──
        _safeTransfer(address(tokenA), address(router), tokenASeed);
        _safeTransfer(address(tokenB), address(router), tokenBSeed);
        console.log("Router token0:", tokenA.balanceOf(address(router)));
        console.log("Router token1:", tokenB.balanceOf(address(router)));

        // ── 10. Initialize pool ──
        PoolKey memory key = PoolKey(
            Currency.wrap(address(tokenA)),
            Currency.wrap(address(tokenB)),
            0x800000, 60, IHooks(hookAddress)
        );
        manager.initialize(key, startSqrtPrice);
        console.log("Pool initialized. tick:", uint256(int256(currentTick)));

        // ── 11. Add liquidity ──
        router.addLiquidity(key, tickLower, tickUpper, int256(uint256(liquidityAmount)));
        console.log("Liquidity added:", liquidityAmount);
        console.log("  tickLower:", uint256(int256(tickLower)));
        console.log("  tickUpper:", uint256(int256(tickUpper)));

        // ── 12. Swap 1: безопасный размер = 1% от token1 стороны пула ──
        uint256 swapAmount1 = tokenBSeed / 100; // 1% от внесённого USDT
        // Конвертируем в WETH эквивалент для zeroForOne свапа
        uint256 swapAmountWETH = (swapAmount1 * 1e18) / (ethPriceUSD8 * 1e10);
        swapAmountWETH = _min(swapAmountWETH, tokenA.balanceOf(address(router)));

        console.log("Swap 1 (WETH->USDT), amount:", swapAmountWETH);
        router.swap(key, true, -int256(swapAmountWETH));
        console.log("Swap 1 done.");
        console.log("  Router WETH after:", tokenA.balanceOf(address(router)));
        console.log("  Router USDT after:", tokenB.balanceOf(address(router)));

        // ── 13. Swap 2: ещё 1% (без искусственного min-порога) ──
        uint256 routerWethBalance = tokenA.balanceOf(address(router));
        uint256 swapAmountWETH2 = _min(swapAmountWETH, routerWethBalance);
        if (swapAmountWETH2 > 0) {
            console.log("Swap 2 (WETH->USDT), amount:", swapAmountWETH2);
            try router.swap(key, true, -int256(swapAmountWETH2)) {
                console.log("Swap 2 done.");
            } catch (bytes memory err) {
                console.log("Swap 2 blocked (PanicMode or limit).");
                console.logBytes(err);
            }
        } else {
            console.log("Swap 2 skipped: zero router WETH balance.");
        }

        // ── 14. Remove liquidity ──
        uint256 bA = tokenA.balanceOf(address(router));
        uint256 bB = tokenB.balanceOf(address(router));
        router.removeLiquidity(key, tickLower, tickUpper, int256(uint256(liquidityAmount)));
        uint256 retA = tokenA.balanceOf(address(router)) - bA;
        uint256 retB = tokenB.balanceOf(address(router)) - bB;
        console.log("Returned token0 (WETH):", retA);
        console.log("Returned token1 (USDT):", retB);
        require(retA > 0 || retB > 0, "FAIL: no tokens returned on remove");
        console.log("PASS: liquidity removed successfully");

        console.log("SUCCESS");
        vm.stopBroadcast();
    }

    function _safeTransfer(address token, address to, uint256 amount) internal {
        (bool ok, bytes memory data) = token.call(
            abi.encodeWithSelector(IERC20Like.transfer.selector, to, amount)
        );
        require(ok && (data.length == 0 || abi.decode(data, (bool))), "TRANSFER_FAILED");
    }

    function _min(uint256 a, uint256 b) internal pure returns (uint256) { return a < b ? a : b; }

    function getTickFromSqrtPrice(uint160 sqrtPriceX96) internal pure returns (int24 tick) {
        int24 lo = -887220; int24 hi = 887220;
        while (lo < hi) {
            int24 mid = lo + (hi - lo + 1) / 2;
            if (getSqrtRatioAtTick(mid) <= sqrtPriceX96) lo = mid; else hi = mid - 1;
        }
        tick = lo;
    }

    function sqrtU256(uint256 x) internal pure returns (uint256 y) {
        if (x == 0) return 0;
        uint256 z = (x + 1) / 2; y = x;
        while (z < y) { y = z; z = (x / z + z) / 2; }
    }

    function getSqrtRatioAtTick(int24 tick) internal pure returns (uint160 sqrtPriceX96) {
        unchecked {
            uint256 absTick = tick < 0 ? uint256(-int256(tick)) : uint256(int256(tick));
            require(absTick <= 887272, "T");
            uint256 ratio = absTick & 0x1 != 0 ? 0xfffcb933bd6fad37aa2d162d1a594001 : 0x100000000000000000000000000000000;
            if (absTick & 0x2     != 0) ratio = (ratio * 0xfff97272373d413259a46990580e213a) >> 128;
            if (absTick & 0x4     != 0) ratio = (ratio * 0xfff2e50f5f656932ef12357cf3c7fdcc) >> 128;
            if (absTick & 0x8     != 0) ratio = (ratio * 0xffe5caca7e10e4e61c3624eaa0941cd0) >> 128;
            if (absTick & 0x10    != 0) ratio = (ratio * 0xffcb9843d60f6159c9db58835c926644) >> 128;
            if (absTick & 0x20    != 0) ratio = (ratio * 0xff973b41fa98c081472e6896dfb254c0) >> 128;
            if (absTick & 0x40    != 0) ratio = (ratio * 0xff2ea16466c96a3843ec78b326b52861) >> 128;
            if (absTick & 0x80    != 0) ratio = (ratio * 0xfe5dee046a99a2a811c461f1969c3053) >> 128;
            if (absTick & 0x100   != 0) ratio = (ratio * 0xfcbe86c7900a88aedcffc83b479aa3a4) >> 128;
            if (absTick & 0x200   != 0) ratio = (ratio * 0xf987a7253ac413176f2b074cf7815e54) >> 128;
            if (absTick & 0x400   != 0) ratio = (ratio * 0xf3392b0822b70005940c7a398e4b70f3) >> 128;
            if (absTick & 0x800   != 0) ratio = (ratio * 0xe7159475a2c29b7443b29c7fa6e889d9) >> 128;
            if (absTick & 0x1000  != 0) ratio = (ratio * 0xd097f3bdfd2022b8845ad8f792aa5825) >> 128;
            if (absTick & 0x2000  != 0) ratio = (ratio * 0xa9f746462d870fdf8a65dc1f90e061e5) >> 128;
            if (absTick & 0x4000  != 0) ratio = (ratio * 0x70d869a156d2a1b890bb3df62baf32f7) >> 128;
            if (absTick & 0x8000  != 0) ratio = (ratio * 0x31be135f97d08fd981231505542fcfa6) >> 128;
            if (absTick & 0x10000 != 0) ratio = (ratio * 0x9aa508b5b7a84e1c677de54f3e99bc9)  >> 128;
            if (absTick & 0x20000 != 0) ratio = (ratio * 0x5d6af8dedb81196699c329225ee604)   >> 128;
            if (absTick & 0x40000 != 0) ratio = (ratio * 0x2216e584f5fa1ea926041bedfe98)      >> 128;
            if (absTick & 0x80000 != 0) ratio = (ratio * 0x48a170391f7dc42444e8fa2)           >> 128;
            if (tick > 0) ratio = type(uint256).max / ratio;
            sqrtPriceX96 = uint160((ratio >> 32) + (ratio % (1 << 32) == 0 ? 0 : 1));
        }
    }
}
